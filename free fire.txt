<script>

let kills = 0;
let hs = 0;

function addKill() {
  kills++;
  update();
}

function addHeadshot() {
  hs++;
  kills++;
  update();
}

function update() {
  document.getElementById("kills").innerText = kills;
  document.getElementById("hs").innerText = hs;

  let percentage = kills === 0 ? 0 : (hs / kills * 100).toFixed(1);

  document.getElementById("percent").innerText =
    percentage + "%";
}

function resetStats() {
  kills = 0;
  hs = 0;
  update();
}

function saveSettings() {
  document.getElementById("saved").innerText =
    "✅ Sensitivity settings saved!";
}

</script>
